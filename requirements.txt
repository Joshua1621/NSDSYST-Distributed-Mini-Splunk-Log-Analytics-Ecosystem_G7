# 1. Use a lightweight, stable version of Python
FROM python:3.11-slim

# 2. Set the working directory inside the container
WORKDIR /app

# 3. Copy the requirements file first to leverage Docker cache
COPY requirements.txt .

# 4. Install the necessary Python dependencies (fastapi, uvicorn, pymongo)
RUN pip install --no-cache-dir -r requirements.txt

# 5. Copy the actual API script into the container
COPY query_api.py .

# 6. Explicitly expose port 8000 so the host machine can route traffic to it
EXPOSE 8000

# 7. Start the Uvicorn ASGI server, binding it to all network interfaces (0.0.0.0)
CMD ["uvicorn", "query_api:app", "--host", "0.0.0.0", "--port", "8000"]
